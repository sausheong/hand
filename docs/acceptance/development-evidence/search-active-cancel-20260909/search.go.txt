package agentio

import (
	"bufio"
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io/fs"
	"os"
	"path"
	"path/filepath"
	"regexp"
	"strings"

	"github.com/go-git/go-git/v5/plumbing/format/gitignore"
	"github.com/sausheong/harness/tool"
)

const defaultMaxSearchResults = 100
const binarySniffLen = 512
const maxSearchLineBytes = 1024 * 1024
const maxSearchOutputBytes = 256 * 1024
const maxIgnoreBytes = 256 * 1024
const maxIgnorePatterns = 10000

// SearchTool uses rooted file access and a Go scanner: Git and rg are not required.
type SearchTool struct{ WorkDir string }
type searchInput struct {
	Path           string `json:"path"`
	NameGlob       string `json:"name_glob"`
	Content        string `json:"content"`
	MaxResults     int    `json:"max_results"`
	MaxBytes       int    `json:"max_bytes"`
	IncludeIgnored bool   `json:"include_ignored"`
	IncludeHidden  bool   `json:"include_hidden"`
}
type searchMatch struct {
	Path string `json:"path"`
	Line int    `json:"line,omitempty"`
	Text string `json:"text,omitempty"`
}
type searchIssue struct {
	Path   string `json:"path"`
	Reason string `json:"reason"`
}

func (t *SearchTool) Name() string { return "search" }
func (t *SearchTool) Description() string {
	return "Find files by basename glob and/or content regex. Returns path:line:text matches and a completeness summary. Honours workspace .gitignore files; ignored/hidden files can be included explicitly. Binary and symlink entries are excluded. Limits, cancellation and scan errors are reported; incomplete results do not establish absence. No shell or rg required."
}
func (t *SearchTool) Parameters() json.RawMessage {
	return json.RawMessage(`{
 "type":"object","properties":{
 "path":{"type":"string","description":"Directory relative to workspace, default ."},
 "name_glob":{"type":"string","description":"Basename glob, e.g. *.go; no multi-segment globs"},
 "content":{"type":"string","description":"Regex to search each text line"},
 "max_results":{"type":"integer","minimum":1,"maximum":10000,"description":"Match cap; default 100"},
 "max_bytes":{"type":"integer","minimum":1024,"maximum":262144,"description":"Output byte budget including status; default 65536"},
 "include_ignored":{"type":"boolean","description":"Include .gitignore-excluded entries and default dependency/state directories"},
 "include_hidden":{"type":"boolean","description":"Include dot files/directories (except .git metadata, always excluded)"}
 }}`)
}
func (t *SearchTool) IsConcurrencySafe(_ json.RawMessage) bool { return true }

func (t *SearchTool) Execute(ctx context.Context, input json.RawMessage) (tool.ToolResult, error) {
	var in searchInput
	fail := func(err error) (tool.ToolResult, error) { return tool.ToolResult{Error: err.Error()}, nil }
	if err := json.Unmarshal(input, &in); err != nil {
		return fail(fmt.Errorf("invalid input: %w", err))
	}
	if in.NameGlob == "" && in.Content == "" {
		return fail(fmt.Errorf("at least one of name_glob or content is required"))
	}
	if _, err := filepath.Match(in.NameGlob, ""); err != nil {
		return fail(fmt.Errorf("invalid name_glob: %w", err))
	}
	if strings.ContainsAny(in.NameGlob, `/\`) {
		return fail(fmt.Errorf("name_glob must match a basename, not a path"))
	}
	if in.MaxResults == 0 {
		in.MaxResults = defaultMaxSearchResults
	}
	if in.MaxResults < 1 || in.MaxResults > 10000 {
		return fail(fmt.Errorf("max_results must be between 1 and 10000"))
	}
	if in.MaxBytes == 0 {
		in.MaxBytes = 65536
	}
	if in.MaxBytes < 1024 || in.MaxBytes > maxSearchOutputBytes {
		return fail(fmt.Errorf("max_bytes must be between 1024 and 262144"))
	}
	var re *regexp.Regexp
	if in.Content != "" {
		var err error
		re, err = regexp.Compile(in.Content)
		if err != nil {
			return fail(fmt.Errorf("invalid content regex: %w", err))
		}
	}
	workspace, err := filepath.Abs(t.WorkDir)
	if err != nil {
		return fail(err)
	}
	workspace, err = filepath.EvalSymlinks(workspace)
	if err != nil {
		return fail(err)
	}
	requested := in.Path
	if requested == "" {
		requested = "."
	}
	requested = resolvePath(workspace, requested)
	if err := tool.ValidatePathInWorkDir(requested, workspace); err != nil {
		return fail(err)
	}
	requested, err = filepath.EvalSymlinks(requested)
	if err != nil {
		return fail(err)
	}
	relative, err := filepath.Rel(workspace, requested)
	if err != nil {
		return fail(err)
	}
	relative = filepath.ToSlash(relative)
	root, err := os.OpenRoot(workspace)
	if err != nil {
		return fail(err)
	}
	defer root.Close()
	info, err := root.Stat(relative)
	if err != nil {
		return fail(err)
	}
	if !info.IsDir() {
		return fail(fmt.Errorf("path %q is not a directory", in.Path))
	}
	matches := []searchMatch{}
	issues := []searchIssue{}
	excluded := map[string]int{}
	var lines []string
	used, issueCount := 0, 0
	truncated, cancelled := false, false
	issue := func(name, reason string) {
		issueCount++
		if len(issues) < 16 {
			issues = append(issues, searchIssue{name, reason})
		}
	}
	patterns := []gitignore.Pattern{}
	loaded := map[string]bool{}
	ignoreBytes := 0
	loadIgnore := func(dir string) {
		if in.IncludeIgnored || loaded[dir] {
			return
		}
		loaded[dir] = true
		name := path.Join(dir, ".gitignore")
		info, e := root.Lstat(name)
		if os.IsNotExist(e) {
			return
		}
		if e != nil {
			issue(name, e.Error())
			return
		}
		if !info.Mode().IsRegular() {
			issue(name, "non-regular ignore file not loaded")
			return
		}
		if info.Size() > int64(maxIgnoreBytes-ignoreBytes) {
			issue(name, "ignore byte budget exceeded")
			return
		}
		f, e := root.Open(name)
		if e != nil {
			issue(name, e.Error())
			return
		}
		defer f.Close()
		scanner := bufio.NewScanner(f)
		scanner.Buffer(make([]byte, 4096), maxIgnoreBytes+1)
		var domain []string
		if dir != "." {
			domain = strings.Split(dir, "/")
		}
		for scanner.Scan() {
			if ctx.Err() != nil {
				return
			}
			text := strings.TrimSuffix(scanner.Text(), "\r")
			ignoreBytes += len(text) + 1
			if ignoreBytes > maxIgnoreBytes || len(patterns) >= maxIgnorePatterns {
				issue(name, "ignore budget exceeded")
				return
			}
			if text == "" || strings.HasPrefix(text, "#") {
				continue
			}
			patterns = append(patterns, gitignore.ParsePattern(text, domain))
		}
		if e := scanner.Err(); e != nil {
			issue(name, e.Error())
		}
	}
	// Load ancestor rules even when the caller starts below the workspace root.
	loadIgnore(".")
	ancestor := "."
	if relative != "." {
		for _, part := range strings.Split(relative, "/") {
			ancestor = path.Join(ancestor, part)
			loadIgnore(ancestor)
		}
	}
	add := func(name string, line int, text string) bool {
		display := filepath.Join(workspace, filepath.FromSlash(name))
		rendered := display
		if line > 0 {
			rendered = fmt.Sprintf("%s:%d:%s", display, line, text)
		}
		// Reserve room for the bounded completeness summary.
		if len(matches) >= in.MaxResults || used+len(rendered)+1 > in.MaxBytes-768 {
			truncated = true
			return false
		}
		matches = append(matches, searchMatch{display, line, text})
		lines = append(lines, rendered)
		used += len(rendered) + 1
		return true
	}
	walkErr := fs.WalkDir(root.FS(), relative, func(name string, d fs.DirEntry, e error) error {
		if ctx.Err() != nil {
			cancelled = true
			return fs.SkipAll
		}
		if e != nil {
			issue(name, e.Error())
			return nil
		}
		skip := func(reason string) error {
			excluded[reason]++
			if d.IsDir() {
				return fs.SkipDir
			}
			return nil
		}
		parts := strings.Split(name, "/")
		// .git is always outside the search scope, including explicit subdirectories.
		for _, part := range parts {
			if part == ".git" {
				return skip("git_metadata")
			}
		}
		if name != relative {
			if !in.IncludeHidden && strings.HasPrefix(d.Name(), ".") {
				return skip("hidden")
			}
			if !in.IncludeIgnored && (d.Name() == "node_modules" || d.Name() == ".hand") {
				return skip("default_ignored")
			}
		}
		if !in.IncludeIgnored && gitignore.NewMatcher(patterns).Match(parts, d.IsDir()) {
			return skip("gitignored")
		}
		if d.IsDir() {
			loadIgnore(name)
			return nil
		}
		if !d.Type().IsRegular() {
			return skip("non_regular")
		}
		if in.NameGlob != "" {
			ok, _ := filepath.Match(in.NameGlob, d.Name())
			if !ok {
				return nil
			}
		}
		if re == nil {
			if !add(name, 0, "") {
				return fs.SkipAll
			}
			return nil
		}
		f, e := root.Open(name)
		if e != nil {
			issue(name, e.Error())
			return nil
		}
		defer f.Close()
		stat, e := f.Stat()
		if e != nil {
			issue(name, e.Error())
			return nil
		}
		if !stat.Mode().IsRegular() {
			return skip("non_regular")
		}
		reader := bufio.NewReaderSize(f, 4096)
		sniff, _ := reader.Peek(binarySniffLen)
		if bytes.IndexByte(sniff, 0) >= 0 {
			return skip("binary")
		}
		scanner := bufio.NewScanner(reader)
		scanner.Buffer(make([]byte, 4096), maxSearchLineBytes)
		number := 0
		for scanner.Scan() {
			if ctx.Err() != nil {
				cancelled = true
				return fs.SkipAll
			}
			number++
			if re.Match(scanner.Bytes()) && !add(name, number, scanner.Text()) {
				return fs.SkipAll
			}
		}
		if e := scanner.Err(); e != nil {
			issue(name, "scan incomplete (line limit 1 MiB): "+e.Error())
		}
		return nil
	})
	if walkErr != nil {
		issue(relative, walkErr.Error())
	}
	if ctx.Err() != nil {
		cancelled = true
	}
	complete := !truncated && !cancelled && issueCount == 0
	summary := fmt.Sprintf("[search complete=%t truncated=%t cancelled=%t issues=%d excluded=%v]", complete, truncated, cancelled, issueCount, excluded)
	if len(lines) == 0 {
		if complete {
			lines = append(lines, "no matches within search scope")
		} else {
			lines = append(lines, "no matches in the portion searched; absence is not established")
		}
	}
	for i := 0; i < len(issues) && i < 2; i++ {
		detail := issues[i].Path + ": " + issues[i].Reason
		if len(detail) > 180 {
			detail = detail[:180] + "..."
		}
		summary += "\n" + detail
	}
	lines = append(lines, summary)
	output := strings.Join(lines, "\n")
	guidance, guidanceComplete := searchInstructions(ctx, workspace, matches, in.MaxBytes-len(output))
	output += guidance
	return tool.ToolResult{Output: output, Metadata: map[string]any{
		"instruction_guidance_complete": guidanceComplete,
		"matches":                       matches, "complete": complete, "truncated": truncated, "cancelled": cancelled,
		"issues": issues, "issue_count": issueCount, "excluded": excluded,
		"include_ignored": in.IncludeIgnored, "include_hidden": in.IncludeHidden,
	}}, nil
}

// Search guidance is scoped to each matched directory. Never truncate instruction
// bodies into misleading partial rules or exceed the caller's output allowance.
func searchInstructions(ctx context.Context, workspace string, matches []searchMatch, available int) (string, bool) {
	const omitted = "\n[Some matched-file guidance was omitted; read each file before acting.]"
	seen := map[string]bool{}
	var out strings.Builder
	complete := true
	for _, match := range matches {
		dir := filepath.Dir(match.Path)
		if seen[dir] {
			continue
		}
		if ctx.Err() != nil || len(seen) >= 128 {
			complete = false
			break
		}
		seen[dir] = true
		report := DiscoverNestedInstructions(workspace, match.Path)
		body := report.Format()
		if body == "" {
			continue
		}
		body = "\nGuidance applicable to files in " + dir + ":" + body
		if out.Len()+len(body)+len(omitted) > available {
			complete = false
			break
		}
		out.WriteString(body)
	}
	if !complete && out.Len()+len(omitted) <= available {
		out.WriteString(omitted)
	}
	return out.String(), complete
}
