name: Release

on:
  push:
    tags:
      - "v*"

permissions:
  contents: write

jobs:
  validate:
    uses: ./.github/workflows/validate.yml
    permissions:
      contents: read

  release:
    needs: validate
    runs-on: ubuntu-latest
    steps:
      - name: Check out code
        uses: actions/checkout@v4

      - name: Set up Go
        uses: actions/setup-go@v5
        with:
          go-version: "1.25.1"

      - name: Build release archives
        run: make dist VERSION=${{ github.ref_name }}

      - name: Verify release archives
        run: python3 scripts/smoke_dist.py --dist dist --version "${{ github.ref_name }}" --commit "$GITHUB_SHA" --require-worker

      - name: Publish GitHub release
        uses: softprops/action-gh-release@v2
        with:
          files: |
            dist/*.tar.gz
            dist/SHA256SUMS
          generate_release_notes: true
