package tui

import (
	"fmt"
	"strings"
	"time"

	"github.com/sausheong/harness/llm"
)

// addUsage returns the field-wise sum of acc and turn. A nil turn (a
// provider that didn't report usage for that call) is a no-op.
func addUsage(acc llm.Usage, turn *llm.Usage) llm.Usage {
	if turn == nil {
		return acc
	}
	acc.InputTokens += turn.InputTokens
	acc.OutputTokens += turn.OutputTokens
	acc.CacheCreationInputTokens += turn.CacheCreationInputTokens
	acc.CacheReadInputTokens += turn.CacheReadInputTokens
	return acc
}

// contextTokens is canonical input from one request. Cache counts are subsets.
func contextTokens(u *llm.Usage) int {
	if u == nil {
		return 0
	}
	return u.InputTokens
}

// totalTokens is every token billed for u: input and output alike.
func totalTokens(u llm.Usage) int {
	return u.InputTokens + u.OutputTokens
}

// formatTokenCount renders n the way the status line and /usage want
// it: compact (one decimal) at or above 1,000, exact below that.
func formatTokenCount(n int) string {
	switch {
	case n >= 1_000_000:
		return trimTrailingZero(fmt.Sprintf("%.1fM", float64(n)/1_000_000))
	case n >= 1_000:
		return trimTrailingZero(fmt.Sprintf("%.1fk", float64(n)/1_000))
	default:
		return fmt.Sprintf("%d", n)
	}
}

// trimTrailingZero drops a redundant ".0" before the unit suffix
// ("12.0k" -> "12k"), so round numbers don't carry a meaningless decimal.
func trimTrailingZero(s string) string {
	return strings.NewReplacer(".0k", "k", ".0M", "M").Replace(s)
}

// formatDuration renders d the way a live stopwatch reads: one decimal
// of sub-minute precision, whole seconds once a turn runs a minute or
// longer (a turn-timer decimal past that point is just noise).
func formatDuration(d time.Duration) string {
	if d < time.Minute {
		return fmt.Sprintf("%.1fs", d.Seconds())
	}
	return d.Round(time.Second).String()
}
